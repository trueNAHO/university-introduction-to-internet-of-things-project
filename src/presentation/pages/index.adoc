[%auto-animate]

[,js,data-id=computeAHPWeights]

function computeAHPWeights(userWeights) {
  for (let i = 0; i < n; i++) {
    matrix[i] = [];

    for (let j = 0; j < n; j++) {
      matrix[i][j] = userWeights[criteria[i]] /
                     userWeights[criteria[j]];
    }
  }
}
----

[%auto-animate]
=== !

[,js,data-id=computeAHPWeights]
----
function computeAHPWeights(userWeights) {
  let geoMeans = [];

  for (let i = 0; i < n; i++) {
    let product = 1;

    for (let j = 0; j < n; j++) {
      product *= matrix[i][j];
    }

    geoMeans[i] = Math.pow(product, 1/n);
  }
}
----

[%auto-animate]
=== !

[,js,data-id=computeAHPWeights]
----
function computeAHPWeights(userWeights) {
  let sumGeo = geoMeans.reduce((sum, val) => sum + val, 0);
  let normalizedWeights = {};

  for (let i = 0; i < n; i++) {
    normalizedWeights[criteria[i]] = geoMeans[i] / sumGeo;
  }
}

function meetsThreshold(room) {

  const co2 = CO2?.co2_level;
  const humidity = Humidity?.humidity;

[%auto-animate]
=== !

[,js,data-id=meetsThreshold]
----
function meetsThreshold(room) {
  return (
    (pm25 !== undefined && pm25 <= 25) &&
    (pm10 !== undefined && pm10 <= 50) &&
    (co2 !== undefined && co2 <= 1000) &&
    (humidity !== undefined &&
      humidity >= 40 && humidity <= 60) &&
    (light !== undefined && light <= 500) &&
    (sound !== undefined && sound <= 70) &&
    (temperature !== undefined &&
      temperature >= 15 && temperature <= 28) &&
    (voc !== undefined && voc <= 1000)
  );
}

async function fetchAndRankRooms() {
  let filteredRooms = roomsData.filter(meetsThreshold);

  // If no room meets thresholds, notify the user.
  if (filteredRooms.length === 0) {
    let list = document.getElementById("room-list");
    list.innerHTML = /* No room available */;
    return;

}

async function fetchAndRankRooms() {
  let minMax = {};
  ["temperature", "humidity", "light", "air_quality"
  ].forEach(key => {
    minMax[key] = {
      min: Math.min(
        ...filteredRooms.map(r => extractValue(r, key))
      ),
      max: Math.max(
        ...filteredRooms.map(r => extractValue(r, key))
      )
    };
  });
}

      humidity: normalize(room, "humidity", minMax),
      light: normalize(room, "light", minMax),
      air_quality: normalize(room, "air_quality", minMax)

  });
}

[,js,data-id=fetchAndRankRooms]
----
async function fetchAndRankRooms() {
  filteredRooms.forEach(room => {
    room.ahpScore =
      room.normalized.temperature *
      criteriaWeights.temperature +
      room.normalized.humidity *
      criteriaWeights.humidity +
      room.normalized.light *
      criteriaWeights.light +
      room.normalized.air_quality *
      criteriaWeights.air_quality;
  });
}
----

[%auto-animate]
=== !

[,js,data-id=fetchAndRankRooms]

async function fetchAndRankRooms() {
  filteredRooms.sort((a, b) => b.ahpScore - a.ahpScore);
}
